# Differences between sake and Windows already understood (tests/winref/compare.py):
#   <probe glob> <regex on the diff line>  # why

# the runner is en-US (LCID 409, console UTF-8) and Windows Server 2025 reports 6.2.9200 to
# un-manifested exes; sake plays a Japanese Windows 7
env-* ^(want|got) +(GetUserDefaultLCID|GetUserDefaultUILanguage|GetConsoleCP|GetVersionEx|GetSystemDefaultLangID|GetUserDefaultLangID|GetSystemDefaultUILanguage)  # runner locale/version
# undefined flags of shifts by more than 1 and of SHLD/SHRD (Intel leaves OF/AF to the CPU)
host_intops-32 ^(want|got) +(shifts|dshift) 
# msvcrt's buffered ftell after a failed fseek reports the buffer position; programs don't rely on it
host_fileseek-32 ^(want|got) +(negative|origin) 
# GDI drawing that matches in shape but not to the pixel: Windows flattens ellipses and Beziers
# into paths (its own flattening), its round-rect/polygon regions round edges its own way, the
# HALFTONE shrink filter is its own, and a region as clip area is kept as its bounding box
gdi-* ^image (shapes|fills_nullpen|polyline_bezier|regions|pens|stretch_halftone|clip):
gdi-* ^(want|got) +GetClipBox 
# 8 and 1 bpp GetDIBits of a 32 bpp bitmap: Windows' nearest-color choice into its default table
gdi-* ^(want|got) +GetDIBits (8|1) 
# sake's console goes out as UTF-8 text (node's stderr): a lone byte >= 0x80 can't come through
crt_printf-* ^(want|got) +%c (-1|-42|255|2147483647) 
# x87 extended precision: msvcrt's sin/cos/tan reduce big arguments its own way (sin(pi) is
# 3.2e-15 there), and pow/_hypot keep 64-bit mantissas inside; koji's x87 is IEEE double
crt_math-32 ^(want|got) +ulp 1 (sin|cos|tan)\((3.14159265358979|1.0000000000000001e\+300|10000000000)\)
crt_math-32 ^(want|got) +ulp 1 (pow\(100.5,3.14159265358979\)|_hypot\(2,100.5\))
# Direct3D 9 on the Windows runner is WARP: its limits and format list are its own (a GPU's
# differ too). sake offers R8G8B8 textures (no driver does: games fall back without them) and
# lacks A2R10G10B10/R32F/V8U8 textures and A8/L8/A8L8/A4R4G4B4 targets so far (TODO).
d3d9-* ^(want|got) +caps vs .* maxtex 
d3d9-* ^(want|got) +caps blend src 
d3d9-* ^(want|got) +format (A4R4G4B4|A8|L8|A8L8|A2R10G10B10|R32F|D32|R8G8B8|V8U8) 
# D3DTADDRESS_BORDER: WebGPU has no border colour; sake clamps (TODO: in the shader)
d3d9-* ^image address:
# StretchRect with D3DTEXF_NONE: WARP filters linearly, GPUs' drivers use point (sake: point)
d3d9-* ^image surfaces:
# Direct3D 11 on the runner is WARP. Its format list and flags are its own; sake has no MSAA
# (quality levels 0: games fall back), no BC6H/BC7, video formats or 3-channel 32-bit textures
# yet (TODO), and creates nothing on other threads concurrently
d3d11-* ^(want|got) +(format |threading |compute on)
# WARP converts floats to UNORM with the spec's 0.6 ulp leeway (17/255 -> 16); GPUs and sake round
d3d11-* ^(want|got) +clear 
# lines and wireframe: WebGPU leaves line rasterization (Bresenham or diamond exit) to the GPU
d3d11-* ^image (lines|wireframe):
# BORDER and MIRROR_ONCE addressing: WebGPU has neither (TODO: in the shader)
d3d11-* ^image address:
# WIC: CMYK -> sRGB goes through a color profile on Windows, 16bppGray -> 8 bits rounds a few
# levels up, and the Linear/Cubic/Fant scalers' weights are their own (sake: plain bilinear).
# TODO: from wicconv's measured tables (cmyk_grid, gray16_ramp, the "w" impulse lines)
imaging-* ^image (gdip_)?(jpeg_cmyk|png_gray16):
imaging-* ^(want|got) +gdip jpeg_cmyk: pixel
imaging-* ^image scale_[123]_
